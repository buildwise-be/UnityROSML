cmake_minimum_required(VERSION 3.8)
project(captured_scene_validation)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
add_executable(validate_captured_scenes validate_captured_scenes.cpp)
target_compile_features(validate_captured_scenes PRIVATE cxx_std_17)
set(VALIDATION_INCLUDE_ROOT /tmp/source/include CACHE PATH "Actual validator headers")
target_include_directories(validate_captured_scenes PRIVATE ${VALIDATION_INCLUDE_ROOT})
ament_target_dependencies(validate_captured_scenes rclcpp moveit_msgs sensor_msgs)
